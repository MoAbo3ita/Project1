👋 أول برنامج لك

دعنا نبدأ ببرنامج بسيط يطبع "Hello, World!":

print("Hello, World!")

هذا كل شيء! Python تجعل الأمور بسيطة للغاية.

📦 المتغيرات

المتغيرات هي حاويات لتخزين البيانات:

# أنواع البيانات الأساسية
name = "Ahmed"      # نص (String)
age = 20            # رقم صحيح (Integer)
height = 1.75       # رقم عشري (Float)
is_student = True   # قيمة منطقية (Boolean)

# طباعة المتغيرات
print("الاسم:", name)
print("العمر:", age)

➕ العمليات الحسابية

# العمليات الأساسية
x = 10
y = 3

print(x + y)   # الجمع: 13
print(x - y)   # الطرح: 7
print(x * y)   # الضرب: 30
print(x / y)   # القسمة: 3.33
print(x // y)  # القسمة الصحيحة: 3
print(x % y)   # باقي القسمة: 1
print(x ** y)  # الأس: 1000

🔀 جمل الشرط

age = 18

if age >= 18:
    print("أنت بالغ")
elif age >= 13:
    print("أنت مراهق")
else:
    print("أنت طفل")

🔄 الحلقات

حلقة for

# طباعة الأرقام من 1 إلى 5
for i in range(1, 6):
    print(i)

# التكرار على قائمة
fruits = ["تفاح", "موز", "برتقال"]
for fruit in fruits:
    print(fruit)

حلقة while

count = 0
while count < 5:
    print(count)
    count += 1

⚡ الدوال

# تعريف دالة
def greet(name):
    return f"مرحباً، {name}!"

# استدعاء الدالة
message = greet("محمد")
print(message)  # مرحباً، محمد!

📋 القوائم

# إنشاء قائمة
numbers = [1, 2, 3, 4, 5]
names = ["أحمد", "محمد", "علي"]

# الوصول للعناصر

names.append("خالد")

# حذف عنصر
names.remove("محمد")

💪 تمرين عملي

اكتب برنامجاً يحسب معدل الطالب:

def calculate_average(grades):
    total = sum(grades)
    average = total / len(grades)
    return average

# اختبار
my_grades = [85, 90, 78, 92]
result = calculate_average(my_grades)
print(f"المعدل: {result:.2f}")
